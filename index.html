// ฟังก์ชันนี้จะถูกเรียกเมื่อหน้าเว็บโหลด เพื่อดึงข้อมูลทั้งหมดไปแสดงผล function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var out = { categories: [], transactions: [], savings: [], debtors: [] }; // 1. ดึงข้อมูลหมวดหมู่ var catSheet = ss.getSheetByName("หมู่หมวด"); if (catSheet) { var catData = catSheet.getDataRange().getValues(); for (var i = 1; i < catData.length; i++) { if (catData[i][1] !== "") { out.categories.push({ id: i, type: catData[i][0] == 'รายรับ' ? 'income' : 'expense', name: catData[i][1], amount: catData[i][2] || 0, isFixed: catData[i][2] ? true : false, emoji: catData[i][3] || '🏷' }); } } } // 2. ดึงข้อมูลรายการย้อนหลัง var transSheet = ss.getSheetByName("รายการ"); if (transSheet) { var transData = transSheet.getDataRange().getValues(); for (var j = 1; j < transData.length; j++) { if (transData[j][0] !== "") { var dateStr = transData[j][0]; if (typeof dateStr === 'object') { dateStr = Utilities.formatDate(dateStr, Session.getScriptTimeZone(), "yyyy-MM-dd"); } out.transactions.push({ id: j, date: dateStr, time: transData[j][1], type: transData[j][2] == 'รายรับ' ? 'income' : 'expense', name: transData[j][3], amount: transData[j][4], note: transData[j][5], emoji: transData[j][6] || '💵' }); } } } // 3. ดึงข้อมูลเงินฝาก (ตามคอลัมน์ A=ชื่อ, B=จำนวนเงิน, C=วันเวลา, D=หมายเหตุ, E=อีโมจิ) var savSheet = ss.getSheetByName("เงินฝาก"); if(savSheet) { var savData = savSheet.getDataRange().getValues(); for (var k = 1; k < savData.length; k++) { if (savData[k][0] !== "") { //[cite: 2] out.savings.push({ name: savData[k][0], amount: savData[k][1], datetime: savData[k][2], note: savData[k][3], emoji: savData[k][4] }); } } } // 4. ดึงข้อมูลคนติดเงิน (ตามคอลัมน์ A=ชื่อ, B=จำนวนเงิน, C=วันเวลา, D=หมายเหตุ, E=อีโมจิ) var debSheet = ss.getSheetByName("คนติดเงิน"); if(debSheet) { var debData = debSheet.getDataRange().getValues(); for (var l = 1; l < debData.length; l++) { if (debData[l][0] !== "") { //[cite: 3] out.debtors.push({ name: debData[l][0], amount: debData[l][1], datetime: debData[l][2], note: debData[l][3], emoji: debData[l][4] }); } } } return ContentService.createTextOutput(JSON.stringify(out)) .setMimeType(ContentService.MimeType.JSON); } // ฟังก์ชันบันทึกข้อมูล function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); if (data.action === "add_trans") { var sheet = ss.getSheetByName("รายการ"); sheet.appendRow([data.date, data.time, data.type, data.name, data.amount, data.note, data.emoji]); } else if (data.action === "add_saving") { // บันทึกลงชีตเงินฝาก[cite: 2] var sheet = ss.getSheetByName("เงินฝาก"); sheet.appendRow([data.name, data.amount, data.datetime, data.note, data.emoji]); } else if (data.action === "add_debtor") { // บันทึกลงชีตคนติดเงิน[cite: 3] var sheet = ss.getSheetByName("คนติดเงิน"); sheet.appendRow([data.name, data.amount, data.datetime, data.note, data.emoji]); } return ContentService.createTextOutput(JSON.stringify({"status": "success"})) .setMimeType(ContentService.MimeType.JSON); }
